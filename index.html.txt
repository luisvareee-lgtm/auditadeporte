<script>
    // ESTADO DEL SISTEMA (Backend simulado)
    let contadoresReportes = {
        'insumos': 0,
        'traslados': 0,
        'becas': 0
    };

    // Función de blanqueo de estilo IA (Simulada con Regex)
    function anonimizarTextoIA(textoCrudo) {
        let textoLimpio = textoCrudo.replace(/(ayer|hoy|lunes|martes|miércoles|jueves|viernes|gustavo|borro|yo|mi silla)/gi, "[DATO PROTEGIDO]");
        return "Reporte verificado en territorio: " + textoLimpio.substring(0, 100) + "... (Estilo neutralizado por IA)";
    }

    function simularEnvioReporte() {
        const rubro = document.getElementById('report-rubro').value;
        const textoCrudo = document.getElementById('report-text').value;
        const tieneEvidencia = document.getElementById('report-evidencia').checked;
        const statusDiv = document.getElementById('report-status');
        
        if(textoCrudo.trim() === "") {
            alert("Por favor escribí un reporte breve.");
            return;
        }

        // 1. Anonimización en memoria RAM
        const textoIA = anonimizarTextoIA(textoCrudo);
        document.getElementById('report-text').value = ''; // Borra el crudo
        
        // 2. Lógica del Umbral Colectivo
        contadoresReportes[rubro]++;
        let cantidadActual = contadoresReportes[rubro];
        
        statusDiv.style.display = 'block';

        if (tieneEvidencia) {
            statusDiv.innerHTML = "<strong>✅ REPORTE VERIFICADO DIRECTAMENTE:</strong> Se adjuntó evidencia documental. Bypass del umbral colectivo aplicado.";
            statusDiv.style.background = "#d1e7dd"; statusDiv.style.color = "#0f5132";
            actualizarTablaPublica(rubro, textoIA);
        } else if (cantidadActual >= 3) {
            statusDiv.innerHTML = `<strong>✅ UMBRAL ALCANZADO (3/3):</strong> El reporte fue validado colectivamente y publicado en la ficha oficial.`;
            statusDiv.style.background = "#d1e7dd"; statusDiv.style.color = "#0f5132";
            actualizarTablaPublica(rubro, textoIA);
        } else {
            statusDiv.innerHTML = `<strong>⏳ EN VERIFICACIÓN (${cantidadActual}/3):</strong> Reporte anonimizado y guardado. Faltan ${3 - cantidadActual} reportes similares del plantel para publicarse en la web.`;
            statusDiv.style.background = "#fff3cd"; statusDiv.style.color = "#856404";
        }
    }

    function actualizarTablaPublica(rubro, textoPublico) {
        const tbody = document.getElementById('tabla-reportes-brown');
        // Actualizamos la fila simulando el impacto en la web pública
        tbody.innerHTML = `
            <tr>
                <td><strong>${rubro.toUpperCase()}</strong></td>
                <td><span class="badge badge-red">Alerta de Faltante</span></td>
                <td style="color: var(--red); font-weight: bold;">${textoPublico}<br><small>Confirmado por Umbral Colectivo / Evidencia</small></td>
            </tr>
        `;
    }

    function mostrarPlantilla(id) {
        document.getElementById('plantilla-a').style.display = 'none';
        document.getElementById('plantilla-b').style.display = 'none';
        document.getElementById(id).style.display = 'block';
    }
</script>
